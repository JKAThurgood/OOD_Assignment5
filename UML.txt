package "SharedStates" {

    abstract class State {
        #context : StateContext
        +consume(character : String)
    }

    class InvalidState {
        +consume(character : String)
    }

    State <|-- InvalidState

    
}

package "Detectors" {

    interface StateContext {
        +changeState(state : State)
    }

    abstract class Detector {
        #currentState : State
        +test(input : String) : Boolean
        +createInitialState() : State
        +changeState(state : State)
    }

    Detector ..|> StateContext

    class IntegerDetector
    class FloatDetector
    class BinaryDetector
    class EmailDetector
    class PasswordDetector

    Detector <|-- IntegerDetector
    Detector <|-- FloatDetector
    Detector <|-- BinaryDetector
    Detector <|-- EmailDetector
    Detector <|-- PasswordDetector

    Detector --> State : owns currentState
    State --> StateContext : changes state through
}
