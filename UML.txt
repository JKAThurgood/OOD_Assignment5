package "Framework" {

    interface State {
        +handle(char: String, context: DetectorContext)
    }

    class DetectorContext {
        -currentState: State
        +setState(state: State)
        +process(char: String)
        +isAccepted(): Boolean
    }

    abstract class Detector {
        +detect(input: String): Boolean
        +createInitialState(): State
    }

    Detector --> DetectorContext
    DetectorContext *--> State : currentState
}

package "Detectors" {

    class IntegerDetector
    class FloatDetector
    class BinaryDetector
    class EmailDetector
    class PasswordDetector

    Detector <|-- IntegerDetector
    Detector <|-- FloatDetector
    Detector <|-- BinaryDetector
    Detector <|-- EmailDetector
    Detector <|-- PasswordDetector
}

package "Integer States" {

    class IntegerStartState
    class IntegerDigitState

    State <|.. IntegerStartState
    State <|.. IntegerDigitState

    IntegerDetector ..> IntegerStartState : initial
    IntegerStartState --> IntegerDigitState
}

package "Float States" {

    class FloatFractionState

    FloatStartState --> FloatWholeState
    FloatStartState --> FloatDotState
    FloatWholeState --> FloatDotState
    FloatDotState --> FloatFractionState
}

package "Binary States" {

    class BinaryStartState
